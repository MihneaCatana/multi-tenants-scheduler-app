# User Management + Observability Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add tenant-admin user management (reset/update/activate/deactivate) and file-based, per-tenant, daily-rotated observability with retention to the Project Simi platform.

**Architecture:** Backend extends the established Fastify controller→service→Drizzle(global) pattern. New tenant-scoped `/users` routes + `/auth/change-password`; the `users` table gains `active` + `must_change_password` columns. Observability uses a single in-process pino routing stream that splits each log line into `logs/{subdomain}/YYYY-MM-DD.json` (or `logs/global/...`) based on the line's `tenantSubdomain` binding, with a daily retention cleanup job. Frontend adds a Users panel, a forced change-password page, and auth-gate logic.

**Tech Stack:** Fastify 5, TypeScript (strict, `verbatimModuleSyntax`, `.js` import extensions), Drizzle ORM + PostgreSQL, pino 10; React 18, Vite, TanStack Query, react-hook-form + Zod, Tailwind.

## Design refinement note (supersedes the spec's "pino-roll" wording)

The approved spec said "pino-roll". Implementation research shows `pino-roll` is a one-file-per-transport tool and cannot route a single logger's stream to per-tenant folders (per-message routing) — and spawning one `pino-roll` worker thread per tenant would be unbounded. To deliver the **exact approved outcome** (one folder per tenant, daily files, rotation, retention) this plan uses a small in-process pino **routing stream** (`src/lib/logs/routing-stream.ts`) that inspects each line's `tenantSubdomain` binding and writes to `logs/{subdomain}/YYYY-MM-DD.json` (or `logs/global/YYYY-MM-DD.json`), rolling over daily. No user-facing behavior changes; this is purely the mechanism.

```ts
import path from 'node:path';
import SonicBoom from 'sonic-boom';
import { Writable } from 'node:stream';

/**
 * A pino DestinationStream that routes each log line to a per-tenant (or global)
 * daily JSON file based on the line's `tenantSubdomain` binding.
 *
 * Layout:
 *   <logDir>/global/YYYY-MM-DD.json
 *   <logDir>/<subdomain>/YYYY-MM-DD.json
 *
 * - One SonicBoom stream per (bucket, day). Buckets are created lazily (mkdir).
 * - Daily rotation: when the UTC date changes, a new file is opened for that
 *   bucket. Stale streams from prior days are closed opportunistically on the
 *   next write that opens a new day for any bucket.
 * - This runs in-process (no worker thread), so it works under tsx without any
 *   thread-stream / structured-clone caveats. SMB-scale write volume makes the
 *   per-line JSON.parse negligible.
 *
 * The stream is a DestinationStream: pino calls `.write(line)` with a string
 * terminated by `\n` (pino's default for custom destinations).
 */
export interface RoutingStreamOptions {
  logDir: string;
}

interface OpenStream {
  stream: SonicBoom;
  bucket: string;
  dateKey: string; // YYYY-MM-DD
}

export class TenantRoutingStream extends Writable {
  private readonly logDir: string;
  // key = `${bucket}/${dateKey}`
  private readonly open = new Map<string, OpenStream>();

  constructor(opts: RoutingStreamOptions) {
    super({ objectMode: false, autoDestroy: true });
    this.logDir = opts.logDir;
  }

  // DestinationStream interface used by pino's multistream.
  write = (chunk: Buffer | string): boolean => {
    this.routeChunk(typeof chunk === 'string' ? chunk : chunk.toString('utf8'));
    return true;
  };

  override _write(_chunk: Buffer, _enc: BufferEncoding, callback: (err?: Error | null) => void): void {
    // Not used in objectMode:false when .write is overridden, but satisfy the
    // Writable contract for any internal piping.
    this.routeChunk(_chunk.toString('utf8'));
    callback();
  }

  private routeChunk(raw: string): void {
    const bucket = this.extractBucket(raw);
    const dateKey = todayUtcKey();
    const key = `${bucket}/${dateKey}`;
    let entry = this.open.get(key);
    if (!entry) {
      entry = this.openStream(bucket, dateKey);
      this.open.set(key, entry);
      // Opportunistic cleanup of streams from previous days (keeps the map
      // bounded to one entry per active bucket).
      this.closeStaleStreams(dateKey);
    }
    entry.stream.write(raw);
  }

  private extractBucket(raw: string): string {
    // Lines always start with '{' (pino JSON). Find tenantSubdomain cheaply via
    // JSON.parse; fall back to 'global' on any parse issue so we never drop a
    // log line.
    try {
      const obj = JSON.parse(raw) as { tenantSubdomain?: string | null };
      if (obj.tenantSubdomain && isSafeBucketName(obj.tenantSubdomain)) {
        return obj.tenantSubdomain;
      }
    } catch {
      /* malformed line — route to global */
    }
    return 'global';
  }

  private openStream(bucket: string, dateKey: string): OpenStream {
    const dir = path.join(this.logDir, bucket);
    const file = path.join(dir, `${dateKey}.json`);
    // SonicBoom mkdir:true creates the directory if missing.
    const stream = new SonicBoom({ dest: file, mkdir: true, sync: false, append: true });
    stream.on('error', () => {
      // Never let a file error kill the process; best-effort logging only.
    });
    return { stream, bucket, dateKey };
  }

  private closeStaleStreams(currentDateKey: string): void {
    for (const [key, entry] of this.open) {
      if (entry.dateKey !== currentDateKey) {
        entry.stream.end();
        this.open.delete(key);
      }
    }
  }

  /** Close every open stream. Call on shutdown. */
  end(): void {
    for (const [, entry] of this.open) {
      entry.stream.end();
    }
    this.open.clear();
    super.end();
  }
}

/** UTC date key as YYYY-MM-DD (matches the daily file granularity). */
export function todayUtcKey(): string {
  return new Date().toISOString().slice(0, 10);
}

/**
 * Restrict the bucket (subdomain) to filesystem-safe characters only. Tenant
 * subdomains are already validated as `[a-z0-9-]` at provisioning time, but we
 * defend in depth against path traversal from a crafted binding.
 */
export function isSafeBucketName(name: string): boolean {
  return /^[a-z0-9][a-z0-9-]{0,62}$/.test(name) && !name.includes('..');
}
```

```ts
import pretty from 'pino-pretty';
import { env } from '../config/env.js';
import { TenantRoutingStream } from './logs/routing-stream.js';

/**
 * Shared pino logger.
 *
 * Every log line flows through `TenantRoutingStream`, which writes it to
 * `LOG_DIR/<subdomain>/YYYY-MM-DD.json` when the line carries a
 * `tenantSubdomain` binding, otherwise to `LOG_DIR/global/YYYY-MM-DD.json`.
 *
 * In development we additionally tee a pino-pretty console stream so the
 * terminal stays readable. Production emits JSON only (file + stdout via the
 * same routing stream; no console prettifier needed).
 */
const routingStream = new TenantRoutingStream({ logDir: env.LOG_DIR });

const redactPaths = [
  'req.headers.authorization',
  'req.headers.cookie',
  '*.password',
  '*.passwordHash',
  '*.refreshToken',
  '*.refreshHash',
  '*.temporaryPassword',
];

const baseConfig = {
  level: env.NODE_ENV === 'test' ? 'silent' : env.LOG_LEVEL,
  base: { service: 'simi-backend' },
  redact: { paths: redactPaths, censor: '[REDACTED]' },
};

const streams: pino.StreamEntry[] = [{ level: baseConfig.level as pino.Level, stream: routingStream }];
if (env.NODE_ENV === 'development') {
  // pino-pretty is a devDep. Static ESM import of its factory returns a
  // writable stream we tee into alongside the routing stream. (See top of file:
  // `import pretty from 'pino-pretty';`)
  streams.push({
    level: baseConfig.level as pino.Level,
    stream: pretty({ colorize: true, translateTime: 'HH:MM:ss' }),
  });
}

export const logger = pino(baseConfig, multistream(streams));

/**
 * Child logger bound to a tenant. Returns (and caches) a child whose every line
 * carries `tenantSubdomain` + `tenantId`, which the routing stream uses to pick
 * the destination folder. `bucketKey` of 'global' returns the root logger.
 */
const tenantLoggerCache = new Map<string, pino.Logger>();
export function getTenantLogger(subdomain: string | null, tenantId: string | null): pino.Logger {
  if (!subdomain || !tenantId) return logger;
  const cached = tenantLoggerCache.get(subdomain);
  if (cached) return cached;
  const child = logger.child({ tenantSubdomain: subdomain, tenantId });
  tenantLoggerCache.set(subdomain, child);
  return child;
}

/** Close all open log file streams — call on shutdown before process exit. */
export function closeLogStreams(): void {
  routingStream.end();
}
```

Create `backend/src/lib/logs/retention.ts` with exactly:

```ts
import { readdir, stat, rm } from 'node:fs/promises';
import path from 'node:path';
import { logger } from '../logger.js';

/**
 * Walk `logDir` recursively and delete files older than `retentionDays` (by
 * mtime). Logs a summary to the global stream. Best-effort: individual file
 * errors are swallowed so one bad file never blocks the sweep.
 */
export async function purgeOldLogs(logDir: string, retentionDays: number): Promise<number> {
  const cutoff = Date.now() - retentionDays * 86_400_000;
  let purged = 0;

  async function walk(dir: string): Promise<void> {
    let entries: import('node:fs').Dirent[];
    try {
      entries = await readdir(dir, { withFileTypes: true });
    } catch {
      return; // dir may not exist yet (no logs written).
    }
    for (const entry of entries) {
      const full = path.join(dir, entry.name);
      if (entry.isDirectory()) {
        await walk(full);
      } else if (entry.isFile()) {
        try {
          const st = await stat(full);
          if (st.mtimeMs < cutoff) {
            await rm(full);
            purged++;
          }
        } catch {
          /* swallow */
        }
      }
    }
  }

  await walk(logDir);
  return purged;
}

let timer: NodeJS.Timeout | null = null;

/**
 * Start the retention job: run once immediately, then every 24h. The job logs
 * its purged-count summary to the global stream. Returns a stop() function.
 */
export function startLogRetentionJob(logDir: string, retentionDays: number): () => void {
  const run = async () => {
    try {
      const purged = await purgeOldLogs(logDir, retentionDays);
      logger.info({ purged, retentionDays }, 'Log retention sweep complete.');
    } catch (err) {
      logger.error({ err }, 'Log retention sweep failed.');
    }
  };
  void run();
  timer = setInterval(() => void run(), 24 * 60 * 60 * 1000);
  return () => {
    if (timer) {
      clearInterval(timer);
      timer = null;
    }
  };
}
```

```ts
import type { FastifyRequest } from 'fastify';

/**
 * Structured audit event. `action` is a dotted, lowercase identifier
 * (e.g. 'user.password_reset'). `target` describes the resource acted upon.
 */
export interface AuditEvent {
  action: string;
  target?: { resource: string; id?: string };
  msg: string;
  level?: 'info' | 'warn';
  /** Extra structured fields merged into the log line. */
  extra?: Record<string, unknown>;
}

/**
 * Emit an audit log line through the request's (tenant-routed) logger. Pulls the
 * actor from `req.userClaims` and the tenant from `req.tenant`. Always uses
 * `req.log` so the line lands in the correct tenant folder (or global/ for
 * apex/bypass requests).
 *
 * Call this EXACTLY ONCE, after the state-changing DB write succeeds.
 */
export function auditLog(req: FastifyRequest, event: AuditEvent): void {
  const claims = req.userClaims;
  const payload: Record<string, unknown> = {
    audit: true,
    action: event.action,
    actor: claims
      ? { userId: claims.sub, role: claims.role }
      : null,
  };
  if (event.target) payload.target = event.target;
  if (event.extra) Object.assign(payload, event.extra);
  const level = event.level ?? 'info';
  req.log[level](payload, event.msg);
}
```

```ts
import { globalDb } from '../../db/client.js';
import { users } from '../../db/schema/global/users.js';
import { hashPassword } from '../../lib/crypto.js';

import { revokeAllSessions } from '../auth/service.js';
import { notFound, forbidden, conflict } from '../../lib/errors.js';
import type { PublicUser } from '../auth/service.js';

function toPublic(u: typeof users.$inferSelect): PublicUser {
  return {
    id: u.id,
    email: u.email,
    role: u.role,
    tenantId: u.tenantId,
    firstName: u.firstName,
    lastName: u.lastName,
    active: u.active,
    mustChangePassword: u.mustChangePassword,
  };

/**
 * Assert the target user belongs to `tenantId`. Throws 404 if not found or in
 * another tenant (never leak cross-tenant existence).
 */
async function loadTenantUser(userId: string, tenantId: string) {
  const [row] = await globalDb
    .select()
    .from(users)
    .where(and(eq(users.id, userId), eq(users.tenantId, tenantId)))
    .limit(1);
  if (!row) throw notFound('User not found.');
  return row;
}

export interface ListUsersInput {
  tenantId: string;
  status?: 'active' | 'inactive';
  limit: number;
  offset: number;
}

export async function listUsers(input: ListUsersInput): Promise<PublicUser[]> {
  const rows = await globalDb
    .select({
      id: users.id,
      email: users.email,
      role: users.role,
      tenantId: users.tenantId,
      firstName: users.firstName,
      lastName: users.lastName,
      active: users.active,
      mustChangePassword: users.mustChangePassword,
      createdAt: users.createdAt,
    })
    .from(users)
    .where(
      and(
        eq(users.tenantId, input.tenantId),
        input.status !== undefined ? eq(users.active, input.status === 'active') : undefined,
      ),
    )
    .limit(input.limit)
    .offset(input.offset)
    .orderBy(users.createdAt);
  return rows.map((r) => ({
    id: r.id,
    email: r.email,
    role: r.role,
    tenantId: r.tenantId,
    firstName: r.firstName,
    lastName: r.lastName,
    active: r.active,
    mustChangePassword: r.mustChangePassword,
  }));
}

export async function updateUser(
  args: {
    tenantId: string;
    targetUserId: string;
    actorUserId: string;
    firstName?: string;
    lastName?: string;
    role?: 'tenant_admin' | 'tenant_user';
  },
): Promise<PublicUser> {
  const target = await loadTenantUser(args.targetUserId, args.tenantId);
  // A tenant admin may not change their own role (prevents self-demotion /
  // escalation edge cases).
  if (args.role !== undefined && args.targetUserId === args.actorUserId) {
    throw forbidden('You cannot change your own role.');
  }
  const set: Record<string, unknown> = { updatedAt: new Date() };
  if (args.firstName !== undefined) set.firstName = args.firstName;
  if (args.lastName !== undefined) set.lastName = args.lastName;
  if (args.role !== undefined) set.role = args.role;

  const [updated] = await globalDb
    .update(users)
    .set(set)
    .where(eq(users.id, target.id))
    .returning();
  return toPublic(updated!);
}

export interface ResetPasswordResult {
  temporaryPassword: string;
}

/**
 * Admin-initiated password reset: generate a random temp password, hash it, set
 * must_change_password, and revoke all of the user's sessions. The temp password
 * is returned ONCE for the admin to communicate out-of-band.
 */
export async function adminResetPassword(
  tenantId: string,
  targetUserId: string,
): Promise<ResetPasswordResult> {
  const target = await loadTenantUser(targetUserId, tenantId);
  // 16 base64url chars (~96 bits of entropy) — strong enough for a temp password
  // the user must change immediately.
  const temporaryPassword = generateToken(12).slice(0, 16);
  const hash = await hashPassword(temporaryPassword);
  await globalDb
    .update(users)
    .set({ passwordHash: hash, mustChangePassword: true, updatedAt: new Date() })
    .where(eq(users.id, target.id));
  await revokeAllSessions(target.id);
  return { temporaryPassword };
}

export async function setUserStatus(
  args: { tenantId: string; targetUserId: string; actorUserId: string; active: boolean },
): Promise<PublicUser> {
  const target = await loadTenantUser(args.targetUserId, args.tenantId);
  // A tenant admin may not deactivate themselves (prevents lockout).
  if (!args.active && args.targetUserId === args.actorUserId) {
    throw forbidden('You cannot deactivate yourself.');
  }
  if (target.active === args.active) {
    throw conflict(`User is already ${args.active ? 'active' : 'inactive'}.`);
  }
  const [updated] = await globalDb
    .update(users)
    .set({ active: args.active, updatedAt: new Date() })
    .where(eq(users.id, target.id))
    .returning();
  if (!args.active) {
    // Stop the deactivated user from refreshing into new access tokens.
    await revokeAllSessions(target.id);
  }
  return toPublic(updated!);
}
```

```tsx
import { useState } from 'react';
import { useForm } from 'react-hook-form';
import { zodResolver } from '@hookform/resolvers/zod';
import { z } from 'zod';
import { api, ApiError } from '../../lib/api';
import { useAuth } from '../../lib/auth';

/**
 * Forced change-password screen. Shown when the logged-in user has
 * mustChangePassword === true. On success the backend revokes other sessions;
 * we sign out locally so the user re-logs-in with their new password (clean
 * session, clears the flag in auth state).
 */
const schema = z
  .object({
    currentPassword: z.string().min(1, 'Required'),
    newPassword: z.string().min(8, 'Min 8 chars').max(128),
    confirm: z.string().min(8, 'Min 8 chars'),
  })
  .refine((d) => d.newPassword === d.confirm, {
    path: ['confirm'],
    message: 'Passwords do not match',
  });
type Values = z.infer<typeof schema>;

export function ChangePasswordPage() {
  const { user, logout } = useAuth();
  const [done, setDone] = useState(false);
  const form = useForm<Values>({ resolver: zodResolver(schema), defaultValues: { currentPassword: '', newPassword: '', confirm: '' } });
  const mutation = form.handleSubmit(async (v) => {
    try {
      await api.changePassword({ currentPassword: v.currentPassword, newPassword: v.newPassword });
      setDone(true);
      // Brief delay so the user reads the confirmation, then sign out.
      setTimeout(() => void logout(), 1500);
    } catch (err) {
      form.setError('currentPassword', {
        message: err instanceof ApiError ? err.message : 'Could not change password.',
      });
    }
  });

  return (
    <div className="flex min-h-screen items-center justify-center bg-slate-50 px-4">
      <div className="w-full max-w-md card p-6">
        <h1 className="mb-1 text-lg font-semibold text-slate-900">Change your password</h1>
        <p className="mb-4 text-sm text-slate-500">
          {user?.email ? `${user.email} — ` : ''}Your password was reset by an administrator. Set a new
          one to continue.
        </p>
        {done ? (
          <div className="rounded-md border border-emerald-200 bg-emerald-50 px-4 py-3 text-sm text-emerald-800">
            Password changed. Signing you out…
          </div>
        ) : (
          <form onSubmit={mutation} className="space-y-4">
            <Field label="Current password" error={form.formState.errors.currentPassword?.message}>
              <input className="input" type="password" {...form.register('currentPassword')} />
            </Field>
            <Field label="New password" error={form.formState.errors.newPassword?.message}>
              <input className="input" type="password" {...form.register('newPassword')} />
            </Field>
            <Field label="Confirm new password" error={form.formState.errors.confirm?.message}>
              <input className="input" type="password" {...form.register('confirm')} />
            </Field>
            <button type="submit" className="btn-primary w-full" disabled={form.formState.isSubmitting}>
              {form.formState.isSubmitting ? 'Changing…' : 'Change password'}
            </button>
          </form>
        )}
      </div>
    </div>
  );
}

function Field({
  label,
  error,
  children,
}: {
  label: string;
  error?: string;
  children: React.ReactNode;
```

```tsx
export function UsersPanel() {

  const [tempPw, setTempPw] = useState<string | null>(null);

  const { data, isLoading, error } = useQuery({
    queryKey: ['users'],
    queryFn: () => api.listUsers({ limit: 100 }),

  const form = useForm<EditValues>({ resolver: zodResolver(editSchema) });

  const updateMutation = useMutation({

            </Field>
            <Field label="Last name" error={form.formState.errors.lastName?.message}>
              <input className="input" {...form.register('lastName')} />
            </Field>
          </div>
          <Field label="Role" error={form.formState.errors.role?.message}>

              {updateMutation.error instanceof ApiError ? updateMutation.error.message : 'Save failed'}
            </p>

        </form>
      </Modal>

          ) : (

              <button type="button" className="btn-secondary" onClick={() => setResetFor(null)}>Cancel</button>
```
